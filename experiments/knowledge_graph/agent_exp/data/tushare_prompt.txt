```python
import os, tushare as ts
ts.set_token(os.getenv("TUSHARE_TOKEN"))
pro = ts.pro_api()

df = pro.daily(
    ts_code="000001.SZ",               # 查询主体：股票代码
    start_date="20260714",              # 时间范围：起始
    end_date="20260714",                # 时间范围：结束
)
if df.empty:
    print("无数据")
else:
    row = df.iloc[0]
    close = row["close"]                # close → DataFrame 列名
    pct_chg = row["pct_chg"]            # pct_chg → DataFrame 列名
    print(f"close={close}, pct_chg={pct_chg}%")
```
